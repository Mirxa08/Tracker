# Tracker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a browser-only site where a person creates day-wise timeline projects of named tracks and events, with images, notes tables, and JSON import/export, deployed to GitHub Pages.

**Architecture:** Four layers — pure `domain` (time maths, schema, transforms), `persistence` (IndexedDB + image handling + transfer), `state` (a pure reducer behind a React context), and `ui` (React components). The domain layer has no React and no browser storage, so the arithmetic that is easy to get wrong is tested directly. Everything below the UI is pure or thinly wrapped, so tests do not need a browser except where the DOM is the subject.

**Tech Stack:** Vite 7, React 19, TypeScript 5 (strict), Vitest 3, React Testing Library, jsdom, `idb` 8, `fake-indexeddb` 6, GitHub Actions.

**Spec:** [`docs/superpowers/specs/2026-11-20-tracker-design.md`](../specs/2026-11-20-tracker-design.md)

## Global Constraints

- Node 20 or newer; npm as the package manager.
- TypeScript `strict: true`. No `any` in committed code; use `unknown` plus narrowing.
- Vite `base: '/Tracker/'` — the site is served from a repository subpath, and a wrong base breaks every asset.
- The event type is named **`TrackEvent`**, never `Event` — `Event` is a DOM global and shadowing it causes confusing type errors.
- Times are ISO 8601 strings in the model (`new Date().toISOString()`), converted to `Date` only at use sites.
- `DAY_WIDTH` is a single exported constant. There are no zoom levels and no per-project width.
- Day arithmetic goes through `startOfDay` / `dayIndex` in `src/domain/time.ts`, never raw millisecond division — a DST day is 23 or 25 hours and raw division is off by one around the transition.
- Runtime dependencies are limited to `react`, `react-dom`, `idb`. Anything else needs a reason recorded in the commit message.
- No network calls at runtime. No analytics, no fonts from a CDN, no telemetry. The site must work offline after first load.
- Dark theme is the default; light is available and `prefers-color-scheme` is respected on first visit.
- Every task ends on a green test run and a commit.

`src/domain/schema.ts`:

```ts
export const SCHEMA_VERSION = 1

export type Theme = 'dark' | 'light' | 'system'
export type Notes = { columns: string[]; rows: string[][] }

export type TrackEvent = {
  id: string
  trackId: string
  name: string
  start: string
  end: string
  color: string
  imageId?: string
  notes?: Notes
}

export type Track = {
  id: string
  name: string
  color: string
  order: number
  collapsed: boolean
}

export type Project = {
  id: string
  name: string
  createdAt: string
  updatedAt: string
  tracks: Track[]
  events: TrackEvent[]
  settings: { theme: Theme }
}

export type ExportFile = {
  schemaVersion: number
  exportedAt: string
  project: Project
  images: Record<string, string>
}

export class ValidationError extends Error {
  field: string
  constructor(field: string, message: string) {
    super(message)
    this.name = 'ValidationError'
    this.field = field
  }
}

function asRecord(value: unknown, field: string): Record<string, unknown> {
  if (typeof value !== 'object' || value === null || Array.isArray(value)) {
    throw new ValidationError(field, `${field} must be an object`)
  }
  return value as Record<string, unknown>
}

function str(source: Record<string, unknown>, field: string, path = field): string {
  const value = source[field]
  if (typeof value !== 'string' || value.trim() === '') {
    throw new ValidationError(field, `${path} must be a non-empty string`)
  }
  return value
}

function iso(source: Record<string, unknown>, field: string, path = field): string {
  const value = str(source, field, path)
  if (Number.isNaN(Date.parse(value))) {
    throw new ValidationError(field, `${path} must be an ISO date string`)
  }
  return value
}

function arr(source: Record<string, unknown>, field: string): unknown[] {
  const value = source[field]
  if (!Array.isArray(value)) throw new ValidationError(field, `${field} must be an array`)
  return value
}

function validateNotes(value: unknown): Notes {
  const raw = asRecord(value, 'notes')
  const columns = arr(raw, 'columns').map((c, i) => {
    if (typeof c !== 'string') throw new ValidationError('notes', `notes column ${i} must be a string`)
    return c
  })
  const rows = arr(raw, 'rows').map((row, i) => {
    if (!Array.isArray(row) || row.some((cell) => typeof cell !== 'string')) {
      throw new ValidationError('notes', `notes row ${i} must be an array of strings`)
    }
    if (row.length !== columns.length) {
      throw new ValidationError('notes', `notes row ${i} has ${row.length} cells but there are ${columns.length} columns`)
    }
    return row as string[]
  })
  return { columns, rows }
}

function validateTrack(value: unknown): Track {
  const raw = asRecord(value, 'track')
  const order = raw.order
  if (typeof order !== 'number' || !Number.isFinite(order)) {
    throw new ValidationError('order', 'track order must be a number')
  }
  return {
    id: str(raw, 'id', 'track id'),
    name: str(raw, 'name', 'track name'),
    color: str(raw, 'color', 'track color'),
    order,
    collapsed: raw.collapsed === true,
  }
}

function validateEvent(value: unknown, trackIds: Set<string>): TrackEvent {
  const raw = asRecord(value, 'event')
  const start = iso(raw, 'start', 'event start')
  const end = iso(raw, 'end', 'event end')
  if (Date.parse(end) <= Date.parse(start)) {
    throw new ValidationError('end', 'event end must be after its start')
  }
  const trackId = str(raw, 'trackId', 'event trackId')
  if (!trackIds.has(trackId)) {
    throw new ValidationError('trackId', `event refers to unknown track "${trackId}"`)
  }
  const event: TrackEvent = {
    id: str(raw, 'id', 'event id'),
    trackId,
    name: str(raw, 'name', 'event name'),
    start,
    end,
    color: str(raw, 'color', 'event color'),
  }
  if (typeof raw.imageId === 'string') event.imageId = raw.imageId
  if (raw.notes !== undefined) event.notes = validateNotes(raw.notes)
  return event
}

export function validateProject(value: unknown): Project {
  const raw = asRecord(value, 'project')
  const settings = asRecord(raw.settings ?? {}, 'settings')
  const theme = settings.theme
  if (theme !== 'dark' && theme !== 'light' && theme !== 'system') {
    throw new ValidationError('theme', 'settings.theme must be dark, light or system')
  }
  const tracks = arr(raw, 'tracks').map(validateTrack)
  const trackIds = new Set(tracks.map((t) => t.id))
  return {
    id: str(raw, 'id', 'project id'),
    name: str(raw, 'name'),
    createdAt: iso(raw, 'createdAt'),
    updatedAt: iso(raw, 'updatedAt'),
    tracks,
    events: arr(raw, 'events').map((e) => validateEvent(e, trackIds)),
    settings: { theme },
  }
}

export function validateExportFile(value: unknown): ExportFile {
  const raw = asRecord(value, 'file')
  const version = raw.schemaVersion
  if (typeof version !== 'number' || !Number.isInteger(version) || version < 1) {
    throw new ValidationError('schemaVersion', 'schemaVersion must be a positive integer')
  }
  if (version > SCHEMA_VERSION) {
    throw new ValidationError(
      'schemaVersion',
      `This file was made by a newer version of Tracker (schema ${version}); this version understands up to ${SCHEMA_VERSION}.`,
    )
  }
  const images = asRecord(raw.images ?? {}, 'images')
  for (const [key, data] of Object.entries(images)) {
    if (typeof data !== 'string') throw new ValidationError('images', `image "${key}" must be a data URL string`)
  }
  return {
    schemaVersion: version,
    exportedAt: iso(raw, 'exportedAt'),
    project: validateProject(raw.project),
    images: images as Record<string, string>,
  }
}

export function migrate(file: ExportFile): ExportFile {
  // Only schema version 1 exists. Future migrations chain here, each step
  // raising schemaVersion by one, so an old file walks forward to current.
  return file
}

export function newId(prefix: string): string {
  return `${prefix}_${crypto.randomUUID().replace(/-/g, '').slice(0, 12)}`
}
```

`src/domain/time.ts`:

```ts
export const DAY_WIDTH = 34
const MS_PER_HOUR = 3_600_000
const MS_PER_DAY = 86_400_000

const MONTHS = [
  'January', 'February', 'March', 'April', 'May', 'June',
  'July', 'August', 'September', 'October', 'November', 'December',
]
const WEEKDAYS = ['Su', 'Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa']

function asDate(value: string | Date): Date {
  return value instanceof Date ? value : new Date(value)
}

export function startOfDay(d: Date): Date {
  return new Date(d.getFullYear(), d.getMonth(), d.getDate())
}

/**
 * Whole calendar days from `from` to `to`. Both are normalised to local
 * midnight and compared as UTC civil dates, so a 23- or 25-hour DST day still
 * counts as exactly one day.
 */
export function dayIndex(from: Date, to: Date): number {
  const a = Date.UTC(from.getFullYear(), from.getMonth(), from.getDate())
  const b = Date.UTC(to.getFullYear(), to.getMonth(), to.getDate())
  return Math.round((b - a) / MS_PER_DAY)
}

function fractionOfDay(d: Date): number {
  return (d.getTime() - startOfDay(d).getTime()) / MS_PER_DAY
}

export function xForTime(t: string | Date, viewStart: Date): number {
  const date = asDate(t)
  return (dayIndex(viewStart, date) + fractionOfDay(date)) * DAY_WIDTH
}

export function timeForX(x: number, viewStart: Date): Date {
  const days = x / DAY_WIDTH
  const whole = Math.floor(days)
  const base = startOfDay(viewStart)
  const day = new Date(base.getFullYear(), base.getMonth(), base.getDate() + whole)
  return new Date(day.getTime() + (days - whole) * MS_PER_DAY)
}

export type DayColumn = {
  date: Date
  x: number
  weekday: string
  day: string
  isToday: boolean
  isWeekend: boolean
}

export function dayColumns(viewStart: Date, viewEnd: Date, now: Date = new Date()): DayColumn[] {
  const count = dayIndex(viewStart, viewEnd) + 1
  const base = startOfDay(viewStart)
  const today = startOfDay(now).getTime()
  const columns: DayColumn[] = []
  for (let i = 0; i < count; i += 1) {
    const date = new Date(base.getFullYear(), base.getMonth(), base.getDate() + i)
    const weekday = date.getDay()
    columns.push({
      date,
      x: i * DAY_WIDTH,
      weekday: WEEKDAYS[weekday],
      day: String(date.getDate()),
      isToday: date.getTime() === today,
      isWeekend: weekday === 0 || weekday === 6,
    })
  }
  return columns
}

export type MonthSpan = { key: string; label: string; x: number; width: number }

export function monthSpans(days: DayColumn[]): MonthSpan[] {
  const spans: MonthSpan[] = []
  for (const column of days) {
    const key = `${column.date.getFullYear()}-${column.date.getMonth()}`
    const last = spans[spans.length - 1]
    if (last && last.key === key) {
      last.width += DAY_WIDTH
    } else {
      spans.push({
        key,
        label: MONTHS[column.date.getMonth()],
        x: column.x,
        width: DAY_WIDTH,
      })
    }
  }
  return spans
}

/**
 * Time left until `end`, as the compact badge shown at a bar's right edge.
 * Returns null once the event has ended.
 */
export function remainingLabel(end: string | Date, now: Date): string | null {
  const ms = asDate(end).getTime() - now.getTime()
  if (ms <= 0) return null
  const minutes = Math.floor(ms / 60_000)
  if (minutes < 60) return `${Math.max(1, minutes)}m`
  const hours = Math.floor(ms / MS_PER_HOUR)
  if (hours < 24) return `${hours}h`
  const days = Math.floor(ms / MS_PER_DAY)
  if (days < 30) return `${days}d`
  return `${Math.floor(days / 30)}mo`
}
```

```ts
import { newId, type Notes, type Project, type TrackEvent } from './schema'
import { nextTrackColor } from './palette'

const MS_PER_DAY = 86_400_000
const PAD_DAYS = 30

function touched(p: Project, changes: Partial<Project>): Project {
  return { ...p, ...changes, updatedAt: new Date().toISOString() }
}

function requireAfter(start: string, end: string): void {
  if (Date.parse(end) <= Date.parse(start)) {
    throw new Error('The end must be after the start.')
  }
}

export function createProject(name: string): Project {
  const now = new Date().toISOString()
  return {
    id: newId('p'),
    name,
    createdAt: now,
    updatedAt: now,
    tracks: [],
    events: [],
    settings: { theme: 'system' },
  }
}

export function addTrack(p: Project, name: string): Project {
  const track = {
    id: newId('t'),
    name,
    color: nextTrackColor(p.tracks.map((t) => t.color)),
    order: p.tracks.length,
    collapsed: false,
  }
  return touched(p, { tracks: [...p.tracks, track] })
}

export function renameTrack(p: Project, trackId: string, name: string): Project {
  return touched(p, {
    tracks: p.tracks.map((t) => (t.id === trackId ? { ...t, name } : t)),
  })
}

export function deleteTrack(p: Project, trackId: string): Project {
  return touched(p, {
    tracks: p.tracks.filter((t) => t.id !== trackId).map((t, i) => ({ ...t, order: i })),
    events: p.events.filter((e) => e.trackId !== trackId),
  })
}

export function reorderTracks(p: Project, orderedIds: string[]): Project {
  const byId = new Map(p.tracks.map((t) => [t.id, t]))
  const tracks = orderedIds
    .map((id) => byId.get(id))
    .filter((t): t is NonNullable<typeof t> => t !== undefined)
    .map((t, i) => ({ ...t, order: i }))
  return touched(p, { tracks })
}

export function addEvent(
  p: Project,
  input: {
    trackId: string
    name: string
    start: string
    end: string
    color?: string
    imageId?: string
    notes?: Notes
  },
): Project {
  const track = p.tracks.find((t) => t.id === input.trackId)
  if (!track) throw new Error(`No such track: ${input.trackId}`)
  requireAfter(input.start, input.end)
  const event: TrackEvent = {
    id: newId('e'),
    trackId: input.trackId,
    name: input.name,
    start: input.start,
    end: input.end,
    color: input.color ?? track.color,
  }
  if (input.imageId) event.imageId = input.imageId
  if (input.notes) event.notes = input.notes
  return touched(p, { events: [...p.events, event] })
}

export function updateEvent(
  p: Project,
  eventId: string,
  patch: Partial<Omit<TrackEvent, 'id'>>,
): Project {
  const existing = p.events.find((e) => e.id === eventId)
  if (!existing) throw new Error(`No such event: ${eventId}`)
  const merged = { ...existing, ...patch }
  requireAfter(merged.start, merged.end)
  if (!p.tracks.some((t) => t.id === merged.trackId)) {
    throw new Error(`No such track: ${merged.trackId}`)
  }
  return touched(p, { events: p.events.map((e) => (e.id === eventId ? merged : e)) })
}

export function deleteEvent(p: Project, eventId: string): Project {
  return touched(p, { events: p.events.filter((e) => e.id !== eventId) })
}

export function eventsForTrack(p: Project, trackId: string): TrackEvent[] {
  return p.events
    .filter((e) => e.trackId === trackId)
    .sort((a, b) => Date.parse(a.start) - Date.parse(b.start))
}

export function projectBounds(p: Project, now: Date): { start: Date; end: Date } {
  const times = p.events.flatMap((e) => [Date.parse(e.start), Date.parse(e.end)])
  const min = times.length ? Math.min(...times, now.getTime()) : now.getTime()
  const max = times.length ? Math.max(...times, now.getTime()) : now.getTime()
  return {
    start: new Date(min - PAD_DAYS * MS_PER_DAY),
    end: new Date(max + PAD_DAYS * MS_PER_DAY),
  }
}
```

```ts
import { openDB, type DBSchema, type IDBPDatabase } from 'idb'
import type { Project } from '../domain/schema'

export type ImageRecord = { id: string; blob: Blob; width: number; height: number }

interface TrackerDB extends DBSchema {
  projects: { key: string; value: Project }
  images: { key: string; value: ImageRecord }
}

const DB_NAME = 'tracker'
const DB_VERSION = 1

export class StorageError extends Error {
  isQuota: boolean
  constructor(message: string, options: { cause?: unknown; isQuota?: boolean } = {}) {
    super(message, { cause: options.cause })
    this.name = 'StorageError'
    this.isQuota = options.isQuota ?? false
  }
}

export function isStorageAvailable(): boolean {
  try {
    return typeof indexedDB !== 'undefined' && indexedDB !== null
  } catch {
    return false
  }
}

let dbPromise: Promise<IDBPDatabase<TrackerDB>> | null = null

export function openDb(): Promise<IDBPDatabase<TrackerDB>> {
  if (!isStorageAvailable()) {
    return Promise.reject(
      new StorageError('This browser has no usable storage, so nothing will be saved.'),
    )
  }
  if (!dbPromise) {
    dbPromise = openDB<TrackerDB>(DB_NAME, DB_VERSION, {
      upgrade(db) {
        if (!db.objectStoreNames.contains('projects')) db.createObjectStore('projects', { keyPath: 'id' })
        if (!db.objectStoreNames.contains('images')) db.createObjectStore('images', { keyPath: 'id' })
      },
    })
  }
  return dbPromise
}

function wrapWriteError(error: unknown): never {
  const isQuota =
    error instanceof DOMException &&
    (error.name === 'QuotaExceededError' || error.name === 'NS_ERROR_DOM_QUOTA_REACHED')
  if (isQuota) {
    throw new StorageError(
      'There is no room left in this browser’s storage. Images are the usual cause — remove a few, or export and delete an old project.',
      { cause: error, isQuota: true },
    )
  }
  throw new StorageError('Saving failed and the change was not applied.', { cause: error })
}

export async function listProjects(): Promise<Project[]> {
  const db = await openDb()
  const all = await db.getAll('projects')
  return all.sort((a, b) => Date.parse(b.updatedAt) - Date.parse(a.updatedAt))
}

export async function getProject(id: string): Promise<Project | undefined> {
  return (await openDb()).get('projects', id)
}

export async function putProject(p: Project): Promise<void> {
  try {
    const db = await openDb()
    await db.put('projects', p)
  } catch (error) {
    wrapWriteError(error)
  }
}

export async function deleteProject(id: string): Promise<void> {
  const db = await openDb()
  await db.delete('projects', id)
}

export async function putImage(record: ImageRecord): Promise<void> {
```

```ts
import { newId } from '../domain/schema'
import { getImage, putImage } from './db'

export const MAX_IMAGE_EDGE = 600

export function fitWithin(
  width: number,
  height: number,
  maxEdge: number,
): { width: number; height: number } {
  const longest = Math.max(width, height)
  if (longest <= maxEdge) return { width, height }
  const scale = maxEdge / longest
  return {
    width: Math.max(1, Math.round(width * scale)),
    height: Math.max(1, Math.round(height * scale)),
  }
}

export function blobToDataUrl(blob: Blob): Promise<string> {
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = () => resolve(String(reader.result))
    reader.onerror = () => reject(reader.error ?? new Error('Could not read the image.'))
    reader.readAsDataURL(blob)
  })
}

export async function dataUrlToBlob(dataUrl: string): Promise<Blob> {
  const match = /^data:([^;,]+)(;base64)?,(.*)$/s.exec(dataUrl)
  if (!match) throw new Error('That is not a data URL.')
  const [, type, base64, payload] = match
  if (!base64) return new Blob([decodeURIComponent(payload)], { type })
  const binary = atob(payload)
  const bytes = new Uint8Array(binary.length)
  for (let i = 0; i < binary.length; i += 1) bytes[i] = binary.charCodeAt(i)
  return new Blob([bytes], { type })
}

/**
 * Downscales to MAX_IMAGE_EDGE and re-encodes to WebP before storing, which
 * keeps the database small without any visible loss at the size a bar shows.
 */
export async function storeImageFile(file: File): Promise<string> {
  const bitmap = await createImageBitmap(file)
  const { width, height } = fitWithin(bitmap.width, bitmap.height, MAX_IMAGE_EDGE)
  const canvas = document.createElement('canvas')
  canvas.width = width
  canvas.height = height
  const context = canvas.getContext('2d')
  if (!context) throw new Error('This browser cannot process images.')
  context.drawImage(bitmap, 0, 0, width, height)
  bitmap.close()

  const blob = await new Promise<Blob | null>((resolve) =>
    canvas.toBlob(resolve, 'image/webp', 0.85),
  )
  if (!blob) throw new Error('Could not convert that image.')

  const id = newId('img')
  await putImage({ id, blob, width, height })
  return id
}
```

```ts
import {
  SCHEMA_VERSION, migrate, newId, validateExportFile,
  type ExportFile, type Project,
} from '../domain/schema'
import { getImage, putImage, putProject } from './db'
import { blobToDataUrl, dataUrlToBlob } from './images'

export async function buildExport(
  project: Project,
  opts: { includeImages: boolean },
): Promise<ExportFile> {
  const images: Record<string, string> = {}
  if (opts.includeImages) {
    const ids = [...new Set(project.events.map((e) => e.imageId).filter((id): id is string => !!id))]
    for (const id of ids) {
      const record = await getImage(id)
      if (record) images[id] = await blobToDataUrl(record.blob)
    }
  }
  return {
    schemaVersion: SCHEMA_VERSION,
    exportedAt: new Date().toISOString(),
    project,
    images,
  }
}

export function serializeExport(file: ExportFile): string {
  return JSON.stringify(file, null, 2)
}

export function exportFileName(project: Project, now: Date): string {
  const slug = project.name
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '') || 'project'
  const date = `${now.getUTCFullYear()}-${String(now.getUTCMonth() + 1).padStart(2, '0')}-${String(now.getUTCDate()).padStart(2, '0')}`
  return `${slug}-${date}.json`
}

/**
 * Import never writes over an existing project: it mints fresh ids for the
 * project, its tracks, its events and its images, so a bad or duplicate file
 * cannot destroy work already in the browser.
 */
export async function importFromJson(json: string): Promise<Project> {
  let parsed: unknown
  try {
    parsed = JSON.parse(json)
  } catch (error) {
    throw new Error('That file could not be read as JSON.', { cause: error })
  }

  const file = migrate(validateExportFile(parsed))

  const imageIdMap = new Map<string, string>()
  for (const [oldId, dataUrl] of Object.entries(file.images)) {
    const blob = await dataUrlToBlob(dataUrl)
    const id = newId('img')
    await putImage({ id, blob, width: 0, height: 0 })
    imageIdMap.set(oldId, id)
  }

  const trackIdMap = new Map<string, string>()
  const tracks = file.project.tracks.map((t) => {
    const id = newId('t')
    trackIdMap.set(t.id, id)
    return { ...t, id }
  })

  const events = file.project.events.map((e) => {
    const mappedImage = e.imageId ? imageIdMap.get(e.imageId) : undefined
    const event = { ...e, id: newId('e'), trackId: trackIdMap.get(e.trackId) ?? e.trackId }
    if (mappedImage) event.imageId = mappedImage
    else delete event.imageId
    return event
  })

  const now = new Date().toISOString()
  const project: Project = {
    ...file.project,
    id: newId('p'),
    tracks,
    events,
    updatedAt: now,
  }
  await putProject(project)
  return project
}
```

```ts
import { deleteEvent, deleteTrack } from '../domain/project'
import type { Project } from '../domain/schema'

export type DialogState =
  | { mode: 'add'; trackId?: string }
  | { mode: 'edit'; eventId: string }
  | null

export type UndoEntry = { label: string; project: Project } | null

export type State = {
  projects: Project[]
  currentId: string | null
  selectedEventId: string | null
  dialog: DialogState
  undo: UndoEntry
  error: string | null
}

export const initialState: State = {
  projects: [],
  currentId: null,
  selectedEventId: null,
  dialog: null,
  undo: null,
  error: null,
}

export type Action =
  | { type: 'loaded'; projects: Project[]; currentId: string | null }
  | { type: 'applyProject'; project: Project }
  | { type: 'selectProject'; projectId: string }
  | { type: 'removeProject'; projectId: string }
  | { type: 'deleteEvent'; eventId: string }
  | { type: 'deleteTrack'; trackId: string }
  | { type: 'undo' }
  | { type: 'openAdd'; trackId?: string }
  | { type: 'openEdit'; eventId: string }
  | { type: 'closeDialog' }
  | { type: 'selectEvent'; eventId: string | null }
  | { type: 'error'; message: string | null }

export function currentProject(state: State): Project | null {
  return state.projects.find((p) => p.id === state.currentId) ?? null
}

function replace(state: State, project: Project): State {
  return {
    ...state,
    projects: state.projects.map((p) => (p.id === project.id ? project : p)),
  }
}

export function reducer(state: State, action: Action): State {
  switch (action.type) {
    case 'loaded':
      return { ...state, projects: action.projects, currentId: action.currentId }

    case 'applyProject':
      return replace(state, action.project)

    case 'selectProject':
      return { ...state, currentId: action.projectId, selectedEventId: null, dialog: null }

    case 'removeProject': {
      const projects = state.projects.filter((p) => p.id !== action.projectId)
      const currentId = state.currentId === action.projectId
        ? (projects[0]?.id ?? null)
        : state.currentId
      return { ...state, projects, currentId, selectedEventId: null, dialog: null }
    }

    case 'deleteEvent': {
      const project = currentProject(state)
      if (!project) return state
      const event = project.events.find((e) => e.id === action.eventId)
      if (!event) return state
      return {
        ...replace(state, deleteEvent(project, action.eventId)),
        undo: { label: `Deleted "${event.name}"`, project },
        selectedEventId: null,
        dialog: null,
      }
    }

    case 'deleteTrack': {
      const project = currentProject(state)
      if (!project) return state
      const track = project.tracks.find((t) => t.id === action.trackId)
      if (!track) return state
      return {
        ...replace(state, deleteTrack(project, action.trackId)),
        undo: { label: `Deleted track "${track.name}"`, project },
        selectedEventId: null,
        dialog: null,
      }
    }

    case 'undo': {
      if (!state.undo) return state
      return { ...replace(state, state.undo.project), undo: null }
    }

    case 'openAdd':
      return { ...state, dialog: { mode: 'add', trackId: action.trackId } }

    case 'openEdit':
      return { ...state, dialog: { mode: 'edit', eventId: action.eventId } }

    case 'closeDialog':
      return { ...state, dialog: null }

    case 'selectEvent':
      return { ...state, selectedEventId: action.eventId }

    case 'error':
      return { ...state, error: action.message }
  }
}
```

```tsx
import { createContext, useContext, useEffect, useReducer, useRef, type ReactNode } from 'react'
import { createProject } from '../domain/project'
import { listProjects, putProject, StorageError } from '../persistence/db'
import { currentProject, initialState, reducer, type Action, type State } from './reducer'

type Store = { state: State; dispatch: (action: Action) => void }

const StoreContext = createContext<Store | null>(null)

const SAVE_DEBOUNCE_MS = 300

export function StoreProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(reducer, initialState)
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null)

  useEffect(() => {
    let cancelled = false
    listProjects()
      .then(async (projects) => {
        if (cancelled) return
        if (projects.length === 0) {
          const first = createProject('My Timeline')
          await putProject(first)
          dispatch({ type: 'loaded', projects: [first], currentId: first.id })
        } else {
          dispatch({ type: 'loaded', projects, currentId: projects[0].id })
        }
      })
      .catch((error: unknown) => {
        const message = error instanceof StorageError
          ? error.message
          : 'Stored projects could not be opened.'
        dispatch({ type: 'error', message })
      })
    return () => { cancelled = true }
  }, [])

  // Writes are debounced so dragging a bar does not hammer IndexedDB.
  const project = currentProject(state)
  useEffect(() => {
    if (!project) return
    if (timer.current) clearTimeout(timer.current)
    timer.current = setTimeout(() => {
      putProject(project).catch((error: unknown) => {
        const message = error instanceof StorageError ? error.message : 'Saving failed.'
        dispatch({ type: 'error', message })
      })
    }, SAVE_DEBOUNCE_MS)
    return () => { if (timer.current) clearTimeout(timer.current) }
  }, [project])

  return <StoreContext.Provider value={{ state, dispatch }}>{children}</StoreContext.Provider>
}

export function useStore(): Store {
  const store = useContext(StoreContext)
  if (!store) throw new Error('useStore must be used inside a StoreProvider')
  return store
}
```

```ts
import { useEffect, useState } from 'react'

/** The current instant, re-rendered on an interval. One second by default. */
export function useNow(intervalMs = 1000): Date {
  const [now, setNow] = useState(() => new Date())
  useEffect(() => {
    const id = setInterval(() => setNow(new Date()), intervalMs)
    return () => clearInterval(id)
  }, [intervalMs])
  return now
}
```

```ts
import { useEffect, useState } from 'react'
import { imageObjectUrl } from '../persistence/images'

/**
 * Resolves a stored image to an object URL, revoking it on unmount so the
 * blob is released. Returns null while loading and when the image is missing,
 * which is expected after a structure-only import.
 */
export function useImageUrl(imageId: string | undefined): string | null {
  const [url, setUrl] = useState<string | null>(null)
  useEffect(() => {
    if (!imageId) {
      setUrl(null)
      return
    }
    let created: string | null = null
    let cancelled = false
    imageObjectUrl(imageId)
      .then((value) => {
        if (cancelled) {
          if (value) URL.revokeObjectURL(value)
          return
        }
        created = value
        setUrl(value)
      })
      .catch(() => setUrl(null))
    return () => {
      cancelled = true
      if (created) URL.revokeObjectURL(created)
    }
  }, [imageId])
  return url
}
```

```ts
function pad(n: number): string {
  return String(n).padStart(2, '0')
}

/** ISO instant -> the `YYYY-MM-DDTHH:mm` a datetime-local input expects, in local time. */
export function toLocalInputValue(iso: string): string {
  if (!iso) return ''
  const d = new Date(iso)
  if (Number.isNaN(d.getTime())) return ''
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`
}

/** The value of a datetime-local input (local time) -> an ISO instant. */
export function fromLocalInputValue(value: string): string {
  if (!value) return ''
  const [date, time] = value.split('T')
  const [y, m, d] = date.split('-').map(Number)
  const [hh, mm] = time.split(':').map(Number)
  return new Date(y, m - 1, d, hh, mm).toISOString()
}
```

```tsx
import type { Notes } from '../domain/schema'

export function NotesTable({ value, onChange }: { value: Notes; onChange: (n: Notes) => void }) {
  const addColumn = () =>
    onChange({
      columns: [...value.columns, `Column ${value.columns.length + 1}`],
      rows: value.rows.map((r) => [...r, '']),
    })

  const addRow = () =>
    onChange({ ...value, rows: [...value.rows, value.columns.map(() => '')] })

  const removeRow = (index: number) =>
    onChange({ ...value, rows: value.rows.filter((_, i) => i !== index) })

  const setColumn = (index: number, text: string) =>
    onChange({ ...value, columns: value.columns.map((c, i) => (i === index ? text : c)) })

  const setCell = (row: number, col: number, text: string) =>
    onChange({
      ...value,
      rows: value.rows.map((r, i) => (i === row ? r.map((c, j) => (j === col ? text : c)) : r)),
    })

  return (
    <div className="notes">
      <table>
        <thead>
          <tr>
            {value.columns.map((column, i) => (
              <th key={i} scope="col">
                <input
                  aria-label={`Column ${i + 1} name`}
                  value={column}
                  onChange={(e) => setColumn(i, e.target.value)}
                />
              </th>
            ))}
            {value.columns.length > 0 && <th scope="col" aria-label="Actions" />}
          </tr>
        </thead>
        <tbody>
          {value.rows.map((row, rowIndex) => (
            <tr key={rowIndex}>
              {row.map((cell, colIndex) => (
                <td key={colIndex}>
                  <input
                    aria-label={`${value.columns[colIndex]} for row ${rowIndex + 1}`}
                    value={cell}
                    onChange={(e) => setCell(rowIndex, colIndex, e.target.value)}
                  />
                </td>
              ))}
              <td>
                <button type="button" onClick={() => removeRow(rowIndex)}>Remove row</button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
      <button type="button" onClick={addColumn}>Add column</button>
      <button type="button" onClick={addRow} disabled={value.columns.length === 0}>Add row</button>
    </div>
  )
```

```tsx
import type { DialogState } from '../state/reducer'
import { NotesTable } from './NotesTable'
import { fromLocalInputValue, toLocalInputValue } from './datetime'
import '../styles/dialog.css'

export type EventDraft = {
  trackId: string
  name: string
  start: string
  end: string
  imageId?: string
  notes?: Notes
}

export function EventDialog({
  project,
  dialog,
  onClose,
  onSubmit,
}: {
  project: Project
  dialog: NonNullable<DialogState>
  onClose: () => void
  onSubmit: (draft: EventDraft & { eventId?: string }) => void
}) {
  const editing = dialog.mode === 'edit'
    ? project.events.find((e) => e.id === dialog.eventId)
    : undefined

  const [name, setName] = useState(editing?.name ?? '')
  const [trackId, setTrackId] = useState(
    editing?.trackId ?? (dialog.mode === 'add' ? dialog.trackId : undefined) ?? project.tracks[0]?.id ?? '',
  )
  const [start, setStart] = useState(toLocalInputValue(editing?.start ?? ''))
  const [end, setEnd] = useState(toLocalInputValue(editing?.end ?? ''))
  const [imageId, setImageId] = useState(editing?.imageId)
  const [notes, setNotes] = useState<Notes>(editing?.notes ?? { columns: [], rows: [] })
  const [error, setError] = useState<string | null>(null)

  async function pickImage(file: File | undefined) {
    if (!file) return
    try {
      setImageId(await storeImageFile(file))
    } catch {
      setError('That image could not be processed.')
    }
  }

  function submit(e: React.FormEvent) {
    e.preventDefault()
    if (!name.trim()) return setError('Give the event a name.')
    if (!trackId) return setError('Create a track first, then add events to it.')
    if (!start || !end) return setError('Both a start and an end are required.')
    const startIso = fromLocalInputValue(start)
    const endIso = fromLocalInputValue(end)
    if (Date.parse(endIso) <= Date.parse(startIso)) return setError('The end must be after the start.')

    const draft: EventDraft & { eventId?: string } = {
      trackId,
      name: name.trim(),
      start: startIso,
      end: endIso,
    }
    if (imageId) draft.imageId = imageId
    if (notes.columns.length > 0) draft.notes = notes
    if (editing) draft.eventId = editing.id
    onSubmit(draft)
  }

  return (
    <div className="dialog__backdrop" role="dialog" aria-modal="true" aria-label={editing ? 'Edit event' : 'Add event'}>
      <form className="dialog" onSubmit={submit}>
        <h2>{editing ? 'Edit event' : 'Add event'}</h2>
        {error && <p role="alert">{error}</p>}

        <label>
          Event name
          <input value={name} onChange={(e) => setName(e.target.value)} />
        </label>

        <label>
          Track
          <select value={trackId} onChange={(e) => setTrackId(e.target.value)}>
            {project.tracks.map((t) => <option key={t.id} value={t.id}>{t.name}</option>)}
          </select>
        </label>

        <label>
          Start
          <input type="datetime-local" value={start} onChange={(e) => setStart(e.target.value)} />
        </label>

        <label>
          End
          <input type="datetime-local" value={end} onChange={(e) => setEnd(e.target.value)} />
        </label>

        <label>
          Image (optional)
          <input type="file" accept="image/*" onChange={(e) => pickImage(e.target.files?.[0])} />
        </label>

        <fieldset>
          <legend>Notes (optional)</legend>
          <NotesTable value={notes} onChange={setNotes} />
        </fieldset>

        <div className="dialog__actions">
          <button type="button" onClick={onClose}>Cancel</button>
          <button type="submit">Save</button>
        </div>
      </form>
    </div>
  )
}
```

```ts
import { useRef, useState } from 'react'
import type { TrackEvent } from '../domain/schema'
import { DAY_WIDTH } from '../domain/time'

export type DragMode = 'move' | 'resize-start' | 'resize-end'

const MS_PER_DAY = 86_400_000

export function snapToDay(date: Date): Date {
  const start = new Date(date.getFullYear(), date.getMonth(), date.getDate())
  const halfway = start.getTime() + MS_PER_DAY / 2
  return date.getTime() >= halfway
    ? new Date(start.getFullYear(), start.getMonth(), start.getDate() + 1)
    : start
}

function shiftDays(iso: string, days: number): Date {
  const d = new Date(iso)
  return new Date(d.getFullYear(), d.getMonth(), d.getDate() + days, d.getHours(), d.getMinutes())
}

export function applyDrag(
  event: TrackEvent,
  mode: DragMode,
  deltaDays: number,
): { start: string; end: string } {
  if (mode === 'move') {
    return {
      start: shiftDays(event.start, deltaDays).toISOString(),
      end: shiftDays(event.end, deltaDays).toISOString(),
    }
  }
  if (mode === 'resize-end') {
    const end = shiftDays(event.end, deltaDays)
    const floor = new Date(Date.parse(event.start) + MS_PER_DAY)
    return { start: event.start, end: (end.getTime() > floor.getTime() ? end : floor).toISOString() }
  }
  const start = shiftDays(event.start, deltaDays)
  const ceiling = new Date(Date.parse(event.end) - MS_PER_DAY)
  return { start: (start.getTime() < ceiling.getTime() ? start : ceiling).toISOString(), end: event.end }
}

export function useBarDrag({
  event,
  onCommit,
}: {
  event: TrackEvent
  onCommit: (dates: { start: string; end: string }) => void
}) {
  const [preview, setPreview] = useState<{ start: string; end: string } | null>(null)
  const origin = useRef(0)

  const onPointerDown = (mode: DragMode) => (e: React.PointerEvent) => {
    e.preventDefault()
    e.stopPropagation()
    origin.current = e.clientX
    const target = e.currentTarget as HTMLElement
    target.setPointerCapture(e.pointerId)

    const move = (ev: PointerEvent) => {
      const deltaDays = Math.round((ev.clientX - origin.current) / DAY_WIDTH)
      setPreview(deltaDays === 0 ? null : applyDrag(event, mode, deltaDays))
    }

    const up = (ev: PointerEvent) => {
      target.releasePointerCapture(ev.pointerId)
      target.removeEventListener('pointermove', move)
      target.removeEventListener('pointerup', up)
      const deltaDays = Math.round((ev.clientX - origin.current) / DAY_WIDTH)
      if (deltaDays !== 0) onCommit(applyDrag(event, mode, deltaDays))
      setPreview(null)
    }

    target.addEventListener('pointermove', move)
    target.addEventListener('pointerup', up)
  }

  return { onPointerDown, preview }
}
```

```ts
export function downloadText(filename: string, text: string): void {
  const blob = new Blob([text], { type: 'application/json' })
  const url = URL.createObjectURL(blob)
  const anchor = document.createElement('a')
  anchor.href = url
  anchor.download = filename
  document.body.appendChild(anchor)
  anchor.click()
  anchor.remove()
  URL.revokeObjectURL(url)
}
```

```tsx
import { useRef, useState } from 'react'
import type { Project } from '../domain/schema'
import '../styles/project-bar.css'

export function ProjectBar({
  projects,
  currentId,
  onSelect,
  onCreate,
  onAddTrack,
  onAddEvent,
  onImport,
  onExport,
  onJumpToNow,
}: {
  projects: Project[]
  currentId: string | null
  onSelect: (id: string) => void
  onCreate: () => void
  onAddTrack: () => void
  onAddEvent: () => void
  onImport: (file: File) => void
  onExport: (opts: { includeImages: boolean }) => void

  const fileInput = useRef<HTMLInputElement>(null)

        Project
        <select value={currentId ?? ''} onChange={(e) => onSelect(e.target.value)}>
          {projects.map((p) => <option key={p.id} value={p.id}>{p.name}</option>)}

      <button type="button" onClick={onJumpToNow}>Now</button>

      <label className="project-bar__images">
        <input
          type="checkbox"
          checked={includeImages}
          onChange={(e) => setIncludeImages(e.target.checked)}
        />
        Include images
      </label>
      <button type="button" onClick={() => onExport({ includeImages })}>Export</button>

      <label className="project-bar__import">
        Import
        <input
          ref={fileInput}
          type="file"
          accept="application/json,.json"
```

- [ ] **Step 1: Write the workflow**

`.github/workflows/deploy.yml`:

```yaml
name: Deploy to GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
      - run: npm ci
      - run: npm test
      - run: npm run build
      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
```

Tests run before the build, so a red suite blocks a deploy rather than publishing a broken site.
